pragma solidity ^0.8.0;

contract Student {
    // Structure to store student details
    struct StudentData {
        string name;
        uint256 rollno;
    }

    // Dynamic array of students
    StudentData[] public studentarr;

    // Function to add a student
    function addStudent(string memory name, uint256 rollno) public {
        // Check whether roll number already exists
        for (uint i = 0; i < studentarr.length; i++) {
            if (studentarr[i].rollno == rollno) {
                revert("Student with this roll number already exists");
            }
        }

        // Add student to array
        studentarr.push(StudentData(name, rollno));
    }

    // Function to get total number of students
    function getstudentLength() public view returns (uint) {
        return studentarr.length;
    }

    // Function to display all students
    function displaystudent() public view returns (StudentData[] memory) {
        return studentarr;
    }

    // Function to get student details by index
    function studentbyindex(uint idx) public view returns (StudentData memory) {
        require(idx < studentarr.length, "Index out of bounds");
        return studentarr[idx];
    }

    // Fallback function
    fallback() external payable {
    }

    // Receive function
    receive() external payable {
    }
}
